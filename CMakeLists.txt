if(DECT2_FFMPEG_PREFIX)
  set(ENV{PKG_CONFIG_PATH} "${DECT2_FFMPEG_PREFIX}/lib/pkgconfig:$ENV{PKG_CONFIG_PATH}")
  pkg_check_modules(FFMPEG_ST REQUIRED libavformat libavcodec libavutil libswscale libswresample)
  add_library(dect2_ffmpeg INTERFACE)
  target_include_directories(dect2_ffmpeg INTERFACE ${DECT2_FFMPEG_PREFIX}/include)
  target_link_directories(dect2_ffmpeg INTERFACE ${DECT2_FFMPEG_PREFIX}/lib)
  # the static libraries and what they need (pkg-config's --static output)
  # pkg-config prints macOS frameworks as "-framework X"; as CMake list items they split into "-framework" and "X", and X would be
  # linked as -lX (ld: library 'CoreFoundation' not found). Keep each pair together as one link option instead.
  set(_ff_libs "")
  set(_ff_fw_next OFF)
  foreach(_ff IN LISTS FFMPEG_ST_STATIC_LDFLAGS)
    if(_ff_fw_next)
      target_link_options(dect2_ffmpeg INTERFACE "SHELL:-framework ${_ff}")
      set(_ff_fw_next OFF)
    elseif(_ff STREQUAL "-framework")
      set(_ff_fw_next ON)
    else()
      list(APPEND _ff_libs "${_ff}")
    endif()
  endforeach()
  target_link_libraries(dect2_ffmpeg INTERFACE ${_ff_libs})
  add_library(PkgConfig::FFMPEG ALIAS dect2_ffmpeg)
  message(STATUS "FFmpeg: static build in ${DECT2_FFMPEG_PREFIX}")
else()
  pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil libswscale libswresample)

# whisper.cpp (MIT, ggml authors): speech to text for the live transcripts (docs/TRANSCRIBE.md). Only the CPU back end of ggml is vendored

option(DECT2_WITH_WHISPER "Speech to text (whisper.cpp) for the live transcripts of airband and marine voice" ON)

  set(BUILD_SHARED_LIBS_ONAIR_SAVED ${BUILD_SHARED_LIBS})

  foreach(_o GGML_METAL GGML_BLAS GGML_OPENMP GGML_CPU_KLEIDIAI GGML_LLAMAFILE GGML_NATIVE GGML_CCACHE GGML_BUILD_TESTS GGML_BUILD_EXAMPLES

  # baseline x86-64 (plus SSE4.2), so those follow DECT2_WIN_AVX2 like the rest of the receiver.
  foreach(_o GGML_AVX GGML_AVX2 GGML_FMA GGML_F16C GGML_BMI2)
    set(${_o} ${DECT2_WIN_AVX2} CACHE BOOL "" FORCE)
  endforeach()

  if(NOT DECT2_WIN_AVX2 AND TARGET ggml-cpu)
    get_target_property(_ggml_cpu_opts ggml-cpu COMPILE_OPTIONS)
    if(_ggml_cpu_opts MATCHES "-mavx|-mfma|-mf16c|-mbmi2|-march=native|/arch:AVX")
      message(FATAL_ERROR "ggml-cpu is built for AVX/AVX2 (${_ggml_cpu_opts}) but DECT2_WIN_AVX2 is off: releases would crash on older CPUs (#226)")
    endif()
  endif()
  set(BUILD_SHARED_LIBS ${BUILD_SHARED_LIBS_ONAIR_SAVED})
  add_library(dect2_whisper STATIC third_party/whisper/src/whisper.cpp third_party/whisper/src/aneforge/whisper-aneforge.cpp)

    target_compile_options(dect2_whisper PRIVATE -w)
  endif()

  target_compile_definitions(dect2core PUBLIC DECT2_HAVE_WHISPER)

  # run, tests/common/golden.h) and fuzz_<mode>_proto a seeded mutation run (meant for DECT2_SANITIZE=ON). Both link the core, which holds the Rust-backed files.
  function(onair_rust_golden_tests MOD)
    add_executable(test_${MOD}_proto_rs tests/${MOD}/test_${MOD}_proto_rs.cpp)
    add_executable(fuzz_${MOD}_proto tests/${MOD}/fuzz_${MOD}_proto.cpp)
    foreach(_t test_${MOD}_proto_rs fuzz_${MOD}_proto)
      target_include_directories(${_t} PRIVATE core/include ${ARGN})
      target_link_libraries(${_t} PRIVATE dect2core)
    endforeach()
    target_compile_options(test_${MOD}_proto_rs PRIVATE -ffp-contract=off)   # the digest must not depend on fused multiply-add in the test's own arithmetic
    add_test(NAME ${MOD}_proto_rs COMMAND test_${MOD}_proto_rs)
    add_test(NAME fuzz_${MOD}_proto COMMAND fuzz_${MOD}_proto)
  endfunction()
